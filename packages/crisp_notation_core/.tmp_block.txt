/// How `layoutStaffSystemSystems` assembles the document into systems.
enum SystemLayoutMode {
  /// The default: break into systems (lines) that fit a target width, then
  /// stack systems top-to-bottom and pack into pages.
  wrapped,

  /// No line breaks — every measure in one continuous system. The resulting
  /// `StaffSystemLayout` has its natural (unbounded) width and is typically
  /// placed inside a horizontal scroll view (the "piano roll" / "卷帘" pattern).
  /// `maxWidth` is ignored and no justification runs.
  singleLine,
}

/// Lays the whole [document] out as **one system containing every measure** —
/// the document is never line-broken. This is the horizontal-scroll ("piano
/// roll" / "卷帘") layout: one row = one `StaffSystem` (possibly multi-staff,
/// e.g. piano + jianpu 弹唱谱), and its natural width can be arbitrarily wide.
///
/// Returns both the laid-out system (for painting) and the raw measure-range
/// info (always 0 → last). Brackets, barline groups and jianpu auto-breaking
/// behave exactly as in `layoutStaffSystem`; the only difference is no wrap.
StaffSystemSystems layoutStaffSystemSystems(
  StaffSystem document,
  LayoutSettings settings, {
  required double maxWidth,
  SystemLayoutMode layoutMode = SystemLayoutMode.wrapped,
  double staffGap = 4.0,
  bool justify = true,
  bool gridAlign = true,
  bool hideEmptyStaves = false,
  Set<int> systemBreaks = const {},
  bool showNoteNames = false,
  bool showNoteOctaves = false,
  NoteNameStyle noteNameStyle = NoteNameStyle.letter,
}) {
  if (layoutMode == SystemLayoutMode.singleLine) {
    // No wrap: the full measure range → one system. Hide-empty still drops
    // globally silent staves on every non-first system; with only one system
    // we keep all parts (the first-system rule), so nothing is dropped.
    final n = document.staves.first.measures.length;
    final layout = layoutStaffSystem(
      document,
      settings,
      staffGap: staffGap,
      gridAlign: gridAlign,
      drawTimeSignature: true,
      finalBarline: true,
      showNoteNames: showNoteNames,
      showNoteOctaves: showNoteOctaves,
      noteNameStyle: noteNameStyle,
    );
    return StaffSystemSystems(
      systems: [StaffSystemSystem(layout: layout, firstMeasure: 0, lastMeasure: n - 1)],
      maxWidth: layout.width,
    );
